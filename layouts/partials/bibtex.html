{{- /* Renders a [bibtex] toggle + hidden BibTeX box for one key. Args (dict): key = entry key in static/bibs/references.bib; id = unique DOM id (a key can appear more than once per page). */ -}} {{- $key := .key -}} {{- $id := .id -}} {{- $content := readFile "static/bibs/references.bib" -}} {{- $pattern := printf `(?s)(@\w+\{%s,.*?\n\})` $key -}} {{- $match := findRE $pattern $content 1 -}} {{- $entry := "" -}} {{- range $match -}}{{- $entry = . -}}{{- end -}} [bibtex]
{{ $entry }}
